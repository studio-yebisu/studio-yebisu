      実行したCLI（原典のまま）
      
      python vendor/train-llm-from-scratch/scripts/prepare_tiny_data.py \
  --dataset text --input input/stories.txt --tokenizer bpe \
  --vocab_size 512 --val_frac 0.125 --out_dir <artifact-dir>/data

python vendor/train-llm-from-scratch/scripts/train_transformer.py \
  --preset tiny --arch modern --device cpu --threads 4 \
  --train-path <artifact-dir>/data/train.h5 --dev-path <artifact-dir>/data/val.h5 \
  --batch-size 16 --lr 0.002 --eval-every 100 --seed 2201 \
  --set context_length=64 --set t_context_length=64 --set t_eval_iters=10 \
  --sample '' --steps 0 --out-path <artifact-dir>/models/baseline.pt

（同じ引数で --steps 800 --out-path <artifact-dir>/models/trained.pt
  --checkpoint-every 200 --checkpoint-dir <artifact-dir>/checkpoints --keep-last 4）

python vendor/train-llm-from-scratch/scripts/generate_text.py \
  --model_path <artifact-dir>/models/trained.pt --input_text "Mira walks to the" \
  --max_new_tokens 48 --top_k 1 --temperature 0.8 --seed 2201 --device cpu
      
      
      run.py はこれらを順に実行し、8つの入力は原典の load_trained_model と model.generate で生成します。P01 は CLI の出力と全文が一致することを確かめています。
